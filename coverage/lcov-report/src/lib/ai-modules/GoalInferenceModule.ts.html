const { primary: PRIMARY_MODEL, fallback: FALLBACK_MODEL } = getDefaultModels();

function toResult(parsed: RawAIResult): AIGoalInferenceResult {
  return {
    inferredGoal: typeof parsed.inferredGoal === 'string' ? parsed.inferredGoal : null,
    confidence: clamp01(parsed.confidence ?? 0),
    rationale: typeof parsed.rationale === 'string' ? parsed.rationale : undefined,
    primaryActionVerb:
      typeof parsed.primaryActionVerb === 'string' && parsed.primaryActionVerb.trim() !== ''
        ? parsed.primaryActionVerb
        : undefined,
  };
}
 
/**
 * Calls LLM to infer the implied goal from a prompt context.
 * Works with both OpenAI (with schema) and Ollama (without schema).
 */
export const aiInferImpliedGoal: AIGoalInferenceFn = async (
  analyzedContext: AnalyzedPromptDataContext
): Promise<AIGoalInferenceResult> => {
  const { promptText, actionVerbDetails, questionDetails, lengthWords } = analyzedContext;
 
  // Terse, domain-aligned instructions for PromptVitals
  const system = [
    'You are a PromptVitals analyst. Infer the single primary user goal from a prompt.',
    'Rules:',
    '- Choose one concise goal. If unclear/insufficient info, set inferredGoal=null and confidence=0.',
    '- Prefer extractive phrasing from the prompt; avoid inventing entities or specifics not present.',
    '- If multiple actions exist, pick the central outcome the user most likely wants.',
    '- Calibrate confidence: 0.9–1 (explicit goal); 0.6–0.8 (strongly implied); 0.3–0.5 (weak signal); <0.3 (unclear).',
    '- Never include extra keys or text; JSON only.',
  ].join('\n');
 
  const user = [
    `Prompt text: ${promptText}`,
    `Action verbs found: ${actionVerbDetails?.verbs?.join(', ') ?? 'none'} (count: ${actionVerbDetails?.count ?? 0})`,
    `Questions found: ${questionDetails?.count ?? 0} (${questionDetails?.types?.join(', ') ?? 'none'})`,
    `Length in words: ${lengthWords ?? 'unknown'}`,
    'Return JSON with fields: inferredGoal (string|null), confidence (0..1), primaryActionVerb (string|null), rationale (string).',
  ].join('\n');
 
  // JSON Schema for structured output (OpenAI only)
  const jsonSchema = {
    name: 'goal_inference_result',
    strict: true,
    schema: {
      type: 'object',
      additionalProperties: false,
      properties: {
        inferredGoal: { description: 'Primary user goal distilled to one short sentence or null', type: ['string', 'null'] },
        confidence: { description: 'Confidence between 0 and 1', type: 'number', minimum: 0, maximum: 1 },
        primaryActionVerb: { description: 'Single dominant action verb or null', type: ['string', 'null'] },
        rationale: { description: '1-2 sentence rationale for the inference', type: 'string' },
      },
      required: ['inferredGoal', 'confidence', 'primaryActionVerb', 'rationale'],
    },
  } as const;
 
  const llm = getLLMClient();
  
  // For Ollama, skip schema; for OpenAI, use schema
  const isOllama = llm.name === 'ollama';
  
  try {
    const { content } = await llm.chatJSON({
      model: PRIMARY_MODEL,
      messages: [
        { role: 'system', content: system },
        { role: 'user', content: user },
      ],
      temperature: 0,
      max_tokens: 200,
      ...(isOllama ? {} : { schema: jsonSchema }),
    });
    const parsed: RawAIResult = JSON.parse(extractJSON(content || '{}'));
    return toResult(parsed);
  } catch (error) {
    console.log('Primary model failed:', error);
    // Fallback: try without schema
    try {
      const { content } = await llm.chatJSON({
        model: FALLBACK_MODEL,
        messages: [
          { role: 'system', content: system },
          { role: 'user', content: user },
        ],
        temperature: 0,
        max_tokens: 200,
      });
      let parsed: RawAIResult = {};
      try {
        parsed = JSON.parse(extractJSON(content || '{}'));
      } catch {
        return { inferredGoal: null, confidence: 0, rationale: 'Invalid JSON from AI' };
      }
      return toResult(parsed);
    } catch (err: any) {
      return { inferredGoal: null, confidence: 0, rationale: 'AI error: ' + (err?.message || 'unknown') };
    }
  }
};
